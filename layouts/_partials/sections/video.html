{{- /* Shows a YouTube video if a link is given; otherwise an uploaded video file, if it exists. Renders nothing when neither is available. */ -}} {{- $s := .s -}} {{- $yt := "" -}} {{- with $s.youtube_url -}} {{- with findRESubmatch `(?:v=|youtu\.be/|embed/|shorts/)([A-Za-z0-9_-]{11})` . 1 -}} {{- $yt = index (index . 0) 1 -}} {{- end -}} {{- end -}} {{- $file := "" -}} {{- with $s.file -}} {{- if fileExists (path.Join "static" .) -}}{{- $file = . -}}{{- end -}} {{- end -}} {{- if or $yt $file }} {{- end }}
